/**
 * Minimal in-process loop model:
 * - input events coalesce into a "latest pointer state"
 * - scheduling is event-driven (no fixed sleeps in the hot path)
 * - overlay update/present is abstracted for real Windows overlay vs synthetic harness
 */
export class AppLoop {
  /**
   * @param {{
   *  overlay: { start:()=>Promise<void>|void, stop:()=>Promise<void>|void, present:(state:any)=>Promise<void>|void },
   *  input: { start:()=>Promise<void>|void, stop:()=>Promise<void>|void, onPointer:(cb:(ev:any)=>void)=>void, onToggleMode?:(cb:()=>void)=>void },
   *  instrumentation: { nowTicks: ()=>bigint, onInput:(ev:any)=>void, onPresent:(ev:any)=>void }
   * }} deps
   */
  constructor({ overlay, input, instrumentation }) {
    this.overlay = overlay;
    this.input = input;
    this.ins = instrumentation;

    this._running = false;
    this._scheduled = false;

    this.state = {
      mode: Modes.Normal,
      x: 0,
      y: 0,
      lastInputTicks: 0n,
      lastPresentTicks: 0n
    };

    this.input.onPointer((ev) => this._handlePointer(ev));
    if (this.input.onToggleMode) {
      this.input.onToggleMode(() => this._handleToggleMode());
    }
  }

  async start() {
    if (this._running) return;
    this._running = true;
    await this.overlay.start();
    await this.input.start();
  }

  async stop() {
    if (!this._running) return;
    this._running = false;
    // stop input first so no further scheduling
    await this.input.stop();
    await this.overlay.stop();
  }

  _handleToggleMode() {
    if (!this._running) return;
    this.state.mode = toggleMode(this.state.mode);
    // force a present so mode visibly updates even without movement
    this._schedulePresent();
  }

  _handlePointer(ev) {
    if (!this._running) return;
    const tInput = this.ins.nowTicks();
    this.state.x = ev.x;
    this.state.y = ev.y;
    this.state.lastInputTicks = tInput;
    this.ins.onInput({ tInput, x: ev.x, y: ev.y, mode: this.state.mode });
    this._schedulePresent();
  }

  _schedulePresent() {
    if (this._scheduled) return;
    this._scheduled = true;
    queueMicrotask(() => {
      this._scheduled = false;
      if (!this._running) return;
      this._presentLatest();
    });
  }

  async _presentLatest() {
    const tPresent = this.ins.nowTicks();
    this.state.lastPresentTicks = tPresent;
    await this.overlay.present({
      x: this.state.x,
      y: this.state.y,
      mode: this.state.mode,
      tInput: this.state.lastInputTicks,
      tPresent
    });
    this.ins.onPresent({
      tInput: this.state.lastInputTicks,
      tPresent,
      x: this.state.x,
      y: this.state.y,
      mode: this.state.mode
    });
  }
}

export class InstrumentationRingBuffer {
  constructor(capacity = 4096) {
    if (!Number.isInteger(capacity) || capacity <= 0) {
      throw new Error('capacity must be a positive integer');
    }
    this.capacity = capacity;
    this._buf = new Array(capacity);
    this._writeIndex = 0;
    this._count = 0;
  }

  record(ev) {
    // ev shape: { kind, tInput?, tPresent?, x?, y?, mode? }
    this._buf[this._writeIndex] = ev;
    this._writeIndex = (this._writeIndex + 1) % this.capacity;
    this._count = Math.min(this.capacity, this._count + 1);
  }

  snapshot() {
    const out = [];
    const start = (this._writeIndex - this._count + this.capacity) % this.capacity;
    for (let i = 0; i < this._count; i++) {
      out.push(this._buf[(start + i) % this.capacity]);
    }
    return out;
  }

  clear() {
    this._writeIndex = 0;
    this._count = 0;
    this._buf = new Array(this.capacity);
  }
}

export function createInstrumentation({ capacity = 4096, nowTicks = defaultNowTicks } = {}) {
  const ring = new InstrumentationRingBuffer(capacity);
  return {
    ring,
    nowTicks,
    onInput(ev) {
      ring.record({ kind: 'input', ...ev });
    },
    onPresent(ev) {
      ring.record({ kind: 'present', ...ev });
    }
  };
}

  const idx = Math.min(sorted.length - 1, Math.max(0, Math.ceil((p / 100) * sorted.length) - 1));

export function summarizeMs(samplesMs) {
  const sorted = [...samplesMs].sort((a, b) => a - b);
  const n = sorted.length;
  if (n === 0) {
    return { n: 0, min: null, max: null, mean: null, p50: null, p95: null };

    min: sorted[0],
    max: sorted[n - 1],

    p95: percentile(sorted, 95)
  };

export async function runWinOverlay({ durationMs = 2000 } = {}) {
  if (process.platform !== 'win32') {
    throw new Error('runWinOverlay is Windows-only');
  }

  const input = createWinInput();
  const overlay = new WinOverlayWindow();
  const ins = createInstrumentation({ capacity: 50_000, nowTicks });
  const loop = new AppLoop({ overlay, input, instrumentation: ins });

  await loop.start();

  // Register a hotkey: Ctrl+Shift+O toggles mode.
  user32.RegisterHotKey(ref.NULL, 1, MOD_CONTROL | MOD_SHIFT, VK_O);

  // Stop after duration.
  setTimeout(() => {
    try {
      // Use our message loop thread to break GetMessage.
      user32.PostQuitMessage(0);
    } catch {
      // ignore
    }
  }, durationMs).unref?.();

  const msg = new MSG();
  // Message loop keeps running regardless of overlay focus.
  while (user32.GetMessageW(msg.ref(), ref.NULL, 0, 0)) {
    const m = msg.message;
    if (m === WM_APP_POINTER) {
      const x = msg.wParam | 0;
      const y = msg.lParam | 0;
      input._emitPointerFromHook(x, y);
    } else if (m === WM_APP_TOGGLE_MODE || m === WM_HOTKEY) {
      input._emitToggleFromHotkey();
    }

    user32.TranslateMessage(msg.ref());
    user32.DispatchMessageW(msg.ref());
  }

  user32.UnregisterHotKey(ref.NULL, 1);
  await loop.stop();

  return ins;
}

function createWinInput() {
  const handlers = { pointer: new Set(), toggle: new Set() };
  const threadId = user32.GetCurrentThreadId();

  const hook = new WinGlobalMouseHook({
    postThreadMessage: (message, wParam, lParam) => user32.PostThreadMessageW(threadId, message, wParam, lParam)
  });

  return {
    onPointer(cb) {
      handlers.pointer.add(cb);
    },
    onToggleMode(cb) {
      handlers.toggle.add(cb);
    },
    async start() {
      hook.start();
    },
    async stop() {
      hook.stop();
    },
    _emitPointerFromHook(x, y) {
      for (const cb of handlers.pointer) cb({ x, y });
    },
    _emitToggleFromHotkey() {
      for (const cb of handlers.toggle) cb();
    }
  };

### Executable behavior (win-hook-overlay.exe)
- Registers a **global low-level mouse hook (WH_MOUSE_LL)** and emits `input_move` events even when unfocused.
- Renders a simple always-on-top overlay (WinForms) that is **click-through** and **does not activate** (WS_EX_TRANSPARENT + WS_EX_NOACTIVATE). A small render loop invalidates only when new input arrives.
- Implements **two runtime-switchable modes** (`normal` / `alt`).
  - Toggle via **Ctrl+Shift+O** (global hotkey) and via test injection flag.
  - Emits `mode` JSON events on every change including `reason`.
- Emits benchmark-friendly **JSONL** to stdout with timestamps:
  - `input_move`: `{ tInput, x, y, source }` where `source` is **"hook"** (WH_MOUSE_LL) or **"inject"** (in-process fallback).

- Added `--os-inject-moves`:
  - Uses Win32 `SendInput(MOUSEEVENTF_MOVE)` to move the OS cursor.
  - Verifies permission by checking `GetCursorPos` delta; if blocked, emits `os_inject_unsupported` with a reason so tests can **skip clearly**.
