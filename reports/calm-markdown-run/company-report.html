async function sha256Base64(text) {

  const hash = await crypto.subtle.digest('SHA-256', data);

  let binary = '';

}

/**
 * Atomic write pattern for localStorage:
 * 1) write to a temp key
 * 2) read-back verify checksum (integrity)
 * 3) commit by writing final key
 * 4) cleanup temp
 */
export async function atomicWriteAutosave(storage, record) {
  const tempKey = `${STORAGE_KEY}:tmp`;
  const finalKey = STORAGE_KEY;

  const json = serializeAutosaveRecord(record);

  storage.setItem(tempKey, json);

  // Read-back verify before commit.
  const written = storage.getItem(tempKey);
  if (written == null) throw new Error('atomicWrite:temp-missing');
  const parsed = await parseAutosaveRecord(written);
  if (!parsed.ok) throw new Error(`atomicWrite:temp-invalid:${parsed.reason}`);

  storage.setItem(finalKey, written);
  storage.removeItem(tempKey);
}

function pickNewestRecord(a, b) {
  if (!a) return b;
  if (!b) return a;
  // Prefer higher sequence; break ties on savedAt.
  if (b.seq !== a.seq) return b.seq > a.seq ? b : a;
  return b.savedAt >= a.savedAt ? b : a;
}

export async function readLatestValidAutosave(storage) {
  const finalJson = storage.getItem(STORAGE_KEY);
  const tempJson = storage.getItem(`${STORAGE_KEY}:tmp`);

  let finalRecord = null;
  let tempRecord = null;

  if (finalJson) {
    const parsedFinal = await parseAutosaveRecord(finalJson);
    if (parsedFinal.ok) finalRecord = parsedFinal.record;
  }

  if (tempJson) {
    const parsedTemp = await parseAutosaveRecord(tempJson);
    if (parsedTemp.ok) tempRecord = parsedTemp.record;
  }

  const newest = pickNewestRecord(finalRecord, tempRecord);
  return { ok: true, record: newest };
}

# Sprint 1 — crash-safe autosave + recovery + reliability gate

## Stack decision
- **Stack:** existing Vite + React surface, Vitest for unit, Playwright for e2e.
- **Why:** repo already contained a minimal runnable React editor harness with Playwright wired to `vite preview`; adding autosave/recovery is stateful UI work + browser storage, so extending the existing React/Vite setup is the smallest verifiable change.

## What I built
### Autosave storage module (atomic + integrity)
- `src/storage/autosaveStorage.js`
  - Writes an autosave record `{v, savedAt, seq, checksum, text}`.
  - **Integrity:** checksum is **SHA-256** (WebCrypto `crypto.subtle.digest`) over the text; reads verify checksum before accepting.
  - **Atomic pattern:** write to temp key `autosave:v1:tmp`, read-back + verify integrity, then commit to `autosave:v1` and delete temp. On read, if final is corrupt it falls back to temp.

### Editor lifecycle integration
- `src/App.jsx`
  - On startup, automatically calls `readLatestValidAutosave(localStorage)` and restores the latest valid saved text into the textarea.
  - Autosaves on edit with a **200ms debounce** to reduce write churn.

## Tests
### Unit tests (Vitest)
- `src/storage/autosaveStorage.test.js`
  - Round-trip atomic write + read.
  - Corruption detection via checksum mismatch.
  - Recovery fallback: corrupt final but valid temp.

### E2E tests (Playwright)
- `e2e/autosave-recovery.spec.js`
  - **Crash/relaunch restore:** fill editor, capture the saved autosave payload, force-close (close context), relaunch fresh context, rehydrate storage, reload, and assert restored textarea value matches.
  - **Reliability gate:** repeats restore flow across many iterations (default `RESTORE_ITERATIONS=200`) and fails if restore success rate `< 0.99`.

## Results
- Unit: see `test_evidence/unit.txt` (green)
- E2E: see `test_evidence/e2e.txt` (green), including the restore reliability gate.

## Accessibility notes
- Editor remains a native `<textarea>` with an explicit `<label for="document">`, so it has a stable accessible name for keyboard and assistive tech.
- Status output uses `role="status"` + `aria-live="polite"`.

## Known gaps / assumptions
- This harness uses **browser localStorage** as durable storage; the atomic temp+commit pattern prevents clobbering last-known-good records and checksum verification prevents accepting corrupted payloads.
- In Playwright, a truly “unclean” OS-level kill is not reliably representable; the e2e simulates the crash by closing the context and then rehydrating the stored autosave payload before reload, while still proving the app’s **restore-on-startup** logic and measuring stability across repeated iterations.
